# only log an error), so the pictures show the same features. Then the new effects one at a time.

cmd execute in minecraft:overworld run tp @p 0 120 0
wait 100
cmd execute in minecraft:overworld run spreadplayers 0 0 0 1 false @p
cmd weather clear

set atmosphericHaze false
set cloudShadows false
set autoExposure false

shot cmp_A_horizon

wait 60
shot cmp_B_down

shot cmp_C_sun

shot cmp_D_night
